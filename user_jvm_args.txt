# Use Generational Z Garbage Collector (ZGC)
-XX:+UseZGC -XX:+ZGenerational

# Set max heap size to ~2GB less than what the system has.
-Xmx15G

# Set min heap size to the same as max, to avoid dealloc/alloc overhead.
-Xms15G

# Pre-allocate the heap before starting.
-XX:+AlwaysPreTouch

# Deduplicate strings on the heap.
-XX:+UseStringDeduplication

# Use custom auth servers.
-Dminecraft.api.env=custom
-Dminecraft.api.auth.host=https://auth.canine.systems/auth
-Dminecraft.api.account.host=https://auth.canine.systems/account
-Dminecraft.api.profiles.host=https://auth.canine.systems/account
-Dminecraft.api.session.host=https://auth.canine.systems/session
-Dminecraft.api.services.host=https://auth.canine.systems/services
