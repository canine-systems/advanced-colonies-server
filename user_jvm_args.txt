# Use Generational Z Garbage Collector (ZGC)
-XX:+UseZGC -XX:+ZGenerational

# Set max heap size to ~7GB less than what the system has.
-Xmx30G

# Set min heap size to the same as max, to avoid dealloc/alloc overhead.
-Xms30G

# Pre-allocate the heap before starting.
-XX:+AlwaysPreTouch

# Deduplicate strings on the heap.
-XX:+UseStringDeduplication

# Expose JMX on port 5000.
-Dcom.sun.management.jmxremote.port=5000
# Disable SSL/TLS for JMX.
-Dcom.sun.management.jmxremote.ssl=false
# Disable authentication for JMX.
-Dcom.sun.management.jmxremote.authenticate=false
